/*verfificador de numeros primos*/
int ehPrimo(int n) {
    int d;
    if (n <= 1) return 0;
    d = 2;
    while (d * d <= n) {
        if ((n - (n / d) * d) == 0) {
            return 0;
        }
        d = d + 1;
    }
    return 1;
}

void main(void) {
    int x;
    x = input();
    output(ehPrimo(x));
}
